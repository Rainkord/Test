313void AuthWidget::onLoginClicked()
314{
315 if (isLocked || m_waitingForAuth) return;
316
317 QString login = loginEdit->text().trimmed();
318 QString pass = passwordEdit->text();
319 if (login.isEmpty() || pass.isEmpty()) {
320 statusLabel->setText(QString::fromUtf8("Заполните все поля."));
321 statusLabel->show();
322 return;
323 }
324
325 QString passHash = QString::fromLatin1(
326 QCryptographicHash::hash(pass.toUtf8(), QCryptographicHash::Sha256).toHex());
327
328 m_pendingLogin = login;
329 m_waitingForAuth = true;
330 loginBtn->setEnabled(false);
331 statusLabel->setStyleSheet(s_hintLabel());
332 statusLabel->setText(QString::fromUtf8("Проверка..."));
333 statusLabel->show();
334
335 ClientSingleton::instance().sendRequestAsync(
336 QString("auth||%1||%2").arg(login, passHash));
337}

363void AuthWidget::onAuthResponseReceived(const QString &response)
364{
365 if (!m_waitingForAuth) return;
366 m_waitingForAuth = false;
367 loginBtn->setEnabled(true);
368
369 if (response.startsWith("auth_code_sent")) {
370 QStringList parts = response.split("||");
371 m_pendingCodeHash = (parts.size() >= 2) ? parts[1].trimmed() : QString();
372 statusLabel->hide();
373 attemptsLabel->hide();
374 emit showVerifyAuth(m_pendingLogin, m_pendingCodeHash);
375 return;
376 }
377
378 if (response == "auth-") {
379 failedAttempts++;
380 if (failedAttempts >= 4) {
381 failedAttempts = 0;
382 lockLevel++;
383 int sec = (lockLevel == 1) ? 30 : (lockLevel == 2 ? 300 : 1800);
384 applyLock(sec, QString::fromUtf8("Превышен лимит попыток. Блокировка на %1 сек.").arg(sec));
385 } else {
386 statusLabel->setStyleSheet(s_errorLabel());
387 statusLabel->setText(QString::fromUtf8("Неверный логин или пароль."));
388 statusLabel->show();
389 attemptsLabel->setText(QString::fromUtf8("Осталось попыток: %1").arg(4 - failedAttempts));
390 attemptsLabel->show();
391 }
392 return;
393 }
394
395 statusLabel->setStyleSheet(s_errorLabel());
396 statusLabel->setText(QString::fromUtf8("Ошибка соединения с сервером."));
397 statusLabel->show();
398}
