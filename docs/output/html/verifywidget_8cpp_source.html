274void VerifyWidget::onVerifyClicked()
275{
276 if (isLocked) return;
277 const QString c = otpInput->code();
278 if (c.length() != 6) return;
279
280 verifyBtn->setEnabled(false);
281 verifyBtn->setStyleSheet(primaryBtnStyle(false));
282
283 const QString entered = QString::fromLatin1(
284 QCryptographicHash::hash(c.toUtf8(), QCryptographicHash::Sha256).toHex());
285
286 if (entered == m_codeHash) {
287 otpInput->setError(false);
288 statusLabel->setStyleSheet(successStyle());
289 statusLabel->setText(QString::fromUtf8("Код верен!"));
290 statusLabel->show();
291 otpInput->setEnabled(false);
292 emit verificationSuccess(m_login);
293 return;
294 }
295
296 // Неверный код
297 otpInput->setError(true);
298 m_attemptsLeft--;
299 if (m_attemptsLeft > 0) {
300 statusLabel->setStyleSheet(errorStyle());
301 statusLabel->setText(
302 QString::fromUtf8("Неверный код. Осталось попыток: %1.").arg(m_attemptsLeft));
303 statusLabel->show();
304 QTimer::singleShot(600, this, [this]() {
305 otpInput->clear();
306 updateVerifyBtn();
307 });
308 } else {
309 applyLock(30, QString::fromUtf8("Превышен лимит попыток. Блокировка на 30 сек."));
310 }
311}
