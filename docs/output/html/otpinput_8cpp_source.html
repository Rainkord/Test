33{

136{

250void OtpInput::fillFromClipboard(const QString &text)
251{
252 QString digits;
253 for (const QChar &c : text)
254 if (c.isDigit()) digits += c;
255 if (digits.isEmpty()) return;
256
257 m_error = false;
258 for (int i = 0; i < N; ++i) {
259 m_boxes[i]->blockSignals(true);
260 if (i < digits.length())
261 m_boxes[i]->setText(QString(digits[i])); // QChar → QString явно
262 else
263 m_boxes[i]->clear();
264 m_boxes[i]->blockSignals(false);
265 updateBoxStyle(i);
266 }
267
268 int focus = qMin(digits.length(), N - 1);
269 m_boxes[focus]->setFocus();
270
271 if (isComplete()) emit completed(code());
272}

274// ── event filter ──────────────────────────────────────────────────────────
286bool OtpInput::eventFilter(QObject *obj, QEvent *ev)
287{
288 int idx = -1;
289 for (int i = 0; i < N; ++i)
290 if (m_boxes[i] == obj) { idx = i; break; }
291 if (idx < 0) return QWidget::eventFilter(obj, ev);
292
293 if (ev->type() == QEvent::FocusIn || ev->type() == QEvent::FocusOut) {
294 updateBoxStyle(idx);
295 return false;
296 }
297
298 if (ev->type() == QEvent::KeyPress) {
299 auto *ke = static_cast<QKeyEvent *>(ev);
300
301 if (ke->key() == Qt::Key_Escape) { emit escPressed(); return true; }
302
303 if (ke->key() == Qt::Key_Backspace) {
304 if (m_boxes[idx]->text().isEmpty()) {
305 if (idx > 0) {
306 m_boxes[idx - 1]->setFocus();
307 m_boxes[idx - 1]->blockSignals(true);
308 m_boxes[idx - 1]->clear();
309 m_boxes[idx - 1]->blockSignals(false);
310 updateBoxStyle(idx - 1);
311 }
312 } else {
313 m_boxes[idx]->blockSignals(true);
314 m_boxes[idx]->clear();
315 m_boxes[idx]->blockSignals(false);
316 updateBoxStyle(idx);
317 }
318 return true;
319 }
320
321 if ((ke->key() == Qt::Key_V && ke->modifiers() & Qt::ControlModifier) ||
322 (ke->key() == Qt::Key_Insert && ke->modifiers() & Qt::ShiftModifier))
323 {
324 fillFromClipboard(QApplication::clipboard()->text());
325 return true;
326 }
327
328 if (ke->key() == Qt::Key_Left && idx > 0) { m_boxes[idx-1]->setFocus(); return true; }
329 if (ke->key() == Qt::Key_Right && idx < N - 1) { m_boxes[idx+1]->setFocus(); return true; }
330
331 if (ke->text().length() == 1 && ke->text()[0].isDigit()) {
332 m_boxes[idx]->blockSignals(true);
333 m_boxes[idx]->setText(ke->text());
334 m_boxes[idx]->blockSignals(false);
335 updateBoxStyle(idx);
336 if (idx < N - 1) {
337 m_boxes[idx + 1]->setFocus();
338 m_boxes[idx + 1]->selectAll();
339 updateBoxStyle(idx + 1);
340 }
341 if (isComplete()) {
342 m_error = false;
343 updateAllStyles();
344 emit completed(code());
345 }
346 return true;
347 }
348 }
349
350 if (ev->type() == QEvent::ContextMenu) {
351 fillFromClipboard(QApplication::clipboard()->text());
352 return true;
353 }
354
355 return QWidget::eventFilter(obj, ev);
356}
