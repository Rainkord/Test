29static QString readEmailTxtValue(const QString &key)
30{
31 // Look for email.txt next to the executable, and also one level up (build dir)
32 QStringList searchPaths = {
33 QCoreApplication::applicationDirPath() + "/email.txt",
34 QCoreApplication::applicationDirPath() + "/../email.txt",
35 QDir::currentPath() + "/email.txt"
36 };
37
38 for (const QString &path : searchPaths) {
39 QFile f(path);
40 if (!f.open(QIODevice::ReadOnly | QIODevice::Text))
41 continue;
42 QTextStream in(&f);
43 while (!in.atEnd()) {
44 QString line = in.readLine();
45 if (line.startsWith('#')) continue; // skip comment lines
46 int sep = line.indexOf('=');
47 if (sep < 0) continue;
48 QString k = line.left(sep).trimmed();
49 QString v = line.mid(sep + 1).trimmed();
50 if (k == key) {
51 v.remove(' '); // strip spaces from value
52 return v;
53 }
54 }
55 }
56 qWarning() << "[SmtpClient] email.txt: key '" << key << "' not found";
57 return {};
58}

115static QString readResponse(QSslSocket *socket, int timeoutMs = 15000)
116{
117 QByteArray data = waitForData(socket, timeoutMs);
118 QString all = QString::fromUtf8(data).trimmed();
119 if (!all.isEmpty()) {
120 QStringList lines = all.split('\n');
121 QString last = lines.last().trimmed();
122 qDebug() << "[SMTP] <--" << last;
123 return last;
124 }
125 qDebug() << "[SMTP] readResponse: no data received";

157static bool doSend(const QString &toEmail,
158 const QString &subjectText,
159 const QString &bodyText)
160{
161 QSslSocket socket;
162 socket.connectToHostEncrypted(SmtpClient::smtpHost, SmtpClient::smtpPort);
163 socket.ignoreSslErrors();
164 if (!socket.waitForEncrypted(20000)) {
165 qDebug() << "[SMTP] SSL connection failed:" << socket.errorString();
166 return false;
167 }
168 qDebug() << "[SMTP] SSL connected";
169
170 QString resp = readResponse(&socket, 15000);
171 if (resp.isEmpty() || !resp.startsWith("220")) return false;
172
173 resp = sendCmd(&socket, "EHLO localhost\r\n");
174 if (!resp.startsWith("250")) return false;
175
176 resp = sendCmd(&socket, "AUTH LOGIN\r\n");
177 if (!resp.startsWith("334")) return false;
178
179 resp = sendCmd(&socket, SmtpClient::senderEmail().toUtf8().toBase64() + "\r\n");
180 if (!resp.startsWith("334")) return false;
181
182 resp = sendCmd(&socket, SmtpClient::senderPassword().toUtf8().toBase64() + "\r\n");
183 if (!resp.startsWith("235")) return false;
184 qDebug() << "[SMTP] Authenticated";
185
186 resp = sendCmd(&socket, "MAIL FROM:<" + SmtpClient::senderEmail().toUtf8() + ">\r\n");
187 if (!resp.startsWith("250")) return false;
188
189 resp = sendCmd(&socket, "RCPT TO:<" + toEmail.toUtf8() + ">\r\n");
190 if (!resp.startsWith("250")) return false;
191
192 resp = sendCmd(&socket, "DATA\r\n");
193 if (!resp.startsWith("354")) return false;
194
195 QByteArray subjectB64 = subjectText.toUtf8().toBase64();
196
197 QByteArray email;
198 email += "Subject: =?UTF-8?B?" + subjectB64 + "?=\r\n";
199 email += "From: " + SmtpClient::senderEmail().toUtf8() + "\r\n";
200 email += "To: " + toEmail.toUtf8() + "\r\n";
201 email += "Content-Type: text/plain; charset=UTF-8\r\n";
202 email += "\r\n";
203 email += bodyText.toUtf8() + "\r\n";
204 email += ".\r\n";
205
206 socket.write(email);
207 socket.flush();
208 socket.waitForBytesWritten(5000);
209 resp = readResponse(&socket);
210 if (!resp.startsWith("250")) {
211 qDebug() << "[SMTP] Email rejected:" << resp;
212 return false;
213 }
214
215 sendCmd(&socket, "QUIT\r\n");
216 socket.disconnectFromHost();
217 if (socket.state() != QAbstractSocket::UnconnectedState)
218 socket.waitForDisconnected(5000);
219
220 qDebug() << "[SMTP] Email sent to" << toEmail;
221 return true;
222}
